function connectDeriv(){ const market = document.getElementById("market").value; const wsURL = "wss://ws.binaryws.com/websockets/v3"; setStatus( "STEP 1/5 — Opening Deriv WebSocket...", "#ffcc66" ); console.log("JM DIAGNOSTIC: Starting connection"); console.log("Endpoint:", wsURL); console.log("Market:", market); try{ socket = new WebSocket(wsURL); }catch(error){ console.error( "STEP 1 FAILED:", error ); setStatus( "STEP 1 FAILED — Browser rejected WebSocket", "#ff5555" ); return; } /* ========================= CONNECTION OPENED ========================= */ socket.onopen = function(){ console.log( "STEP 1 SUCCESS — WebSocket OPEN" ); setStatus( "STEP 2/5 — WebSocket connected", "#44dd88" ); /* Request server time. This is a simple connectivity confirmation. */ socket.send( JSON.stringify({ time: 1, req_id: 100 }) ); /* Ask Deriv which symbols are available. */ socket.send( JSON.stringify({ active_symbols: "brief", product_type: "basic", req_id: 101 }) ); /* Subscribe to the selected live market. */ socket.send( JSON.stringify({ ticks: market, subscribe: 1, req_id: 102 }) ); /* Request recent tick history. This also proves the symbol is available. */ socket.send( JSON.stringify({ ticks_history: market, count: 20, end: "latest", style: "ticks", req_id: 103 }) ); }; /* ========================= MESSAGES ========================= */ socket.onmessage = function(event){ try{ const data = JSON.parse(event.data); console.log( "DERIV RESPONSE:", data ); /* SERVER TIME */ if( data.req_id === 100 && data.msg_type === "time" ){ setStatus( "STEP 3/5 — Deriv server responding", "#44dd88" ); console.log( "STEP 3 SUCCESS — Server time:", data.time ); } /* ACTIVE SYMBOLS */ if( data.req_id === 101 && data.msg_type === "active_symbols" ){ const found = data.active_symbols.some( symbol => symbol.symbol === market ); if(found){ console.log( "STEP 4 SUCCESS — Market exists:", market ); setStatus( "STEP 4/5 — Market confirmed: " + market, "#44dd88" ); }else{ console.error( "MARKET NOT FOUND:", market ); setStatus( "Market not found: " + market, "#ff5555" ); } } /* TICK HISTORY */ if( data.req_id === 103 && data.msg_type === "history" ){ console.log( "HISTORY RECEIVED:", data ); setStatus( "STEP 4/5 — History received; waiting for live tick", "#44dd88" ); } /* LIVE TICK */ if( data.msg_type === "tick" && data.tick ){ const quote = data.tick.quote; const symbol = data.tick.symbol; const epoch = data.tick.epoch; console.log( "STEP 5 SUCCESS — LIVE TICK", { symbol:symbol, quote:quote, epoch:epoch } ); const digit = extractLastDigit(quote); if(digit !== null){ addDigit(digit); setStatus( "LIVE — " + symbol + " • Digit " + digit, "#44dd88" ); }else{ setStatus( "Tick received but digit extraction failed", "#ff5555" ); } } /* DERIV ERROR */ if(data.error){ console.error( "DERIV API ERROR:", data.error ); setStatus( "DERIV ERROR: " + ( data.error.message || data.error.code || "Unknown error" ), "#ff5555" ); } }catch(error){ console.error( "MESSAGE PROCESSING ERROR:", error ); setStatus( "Message processing error", "#ff5555" ); } }; /* ========================= WEBSOCKET ERROR ========================= */ socket.onerror = function(error){ console.error( "WEBSOCKET ERROR:", error ); setStatus( "WEBSOCKET ERROR — check console", "#ff5555" ); }; /* ========================= WEBSOCKET CLOSED ========================= */ socket.onclose = function(event){ socket = null; console.error( "WEBSOCKET CLOSED", { code:event.code, reason:event.reason, wasClean:event.wasClean } ); if(running){ setStatus( "CLOSED — code " + event.code + " — reconnecting...", "#ffcc66" ); reconnectTimer = setTimeout( connectDeriv, 5000 ); }else{ setStatus( "Disconnected — code " + event.code, "#ff5555" ); } }; }
